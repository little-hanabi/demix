// mask_spectral.cpp — standalone alternative to Audio::mask, for evaluation only.
// Not wired into the build; nothing in C++/ references it.
//
// Idea: instead of running one ISTFT per chunk and crossfading the two time-domain
// reconstructions over L_MIX samples, crossfade the per-chunk complex masks over the
// M overlapping FRAMES, then reconstruct with a single continuous ISTFT over all T_
// frames of the file. Per sample, the divisor is the envelope of the frames actually
// summed, so the file head AND tail are normalized exactly (the current code's
// under-scaled last ~N_FFT/2 samples disappear).
//
// Inputs map 1:1 onto the existing Audio members / Model output:
//   spec : whole-file STFT, complex, index (c * T_ + t) * N_BIN + f
//   buf  : Model::infer output, chunk n at n * N_BUF, frame layout ((t*N_BIN+f)*C+c)*2
//   wav  : C channels, each resized to L_ beforehand (as write() already does);
//          every sample in [0, L_) is assigned exactly once, so pre-zeroing is not needed
//
// Scratch: one N_BIN complex + one N_FFT real FFT buffer, plus two rolling N_FFT
// float windows (~33 KB total) — independent of file length and chunk count.
// The chunk blend itself needs no buffer: buf already holds every chunk's mask,
// so the overlap reads chunk n-1 and chunk n in place.
//
// FFT count is C * T_ single 2048-point c2r transforms through one reused plan —
// fewer transforms than the current N_ * C * T batch, since overlapping frames are
// no longer synthesized twice, and the N_*C*T*N_FFT-sized temporaries are gone.

namespace demix {
    namespace {
        // identical to the (fixed) window in audio.cpp; reuse that one when integrating
        const auto WIN = [] {
            constexpr float PI = std::numbers::pi_v<float>;
            std::array<float, N_FFT> w{};
            for (int i = 0; i < N_FFT; ++i) w[i] = 0.5f * (1.0f - std::cos(2.0f * PI * i / N_FFT));
            return w;
        }();
    }

    void mask_spectral(const fftwf_complex* spec, const std::span<const float> buf,
                       const uint64_t T_, const uint64_t L_, const uint64_t N_,
                       std::vector<std::vector<float>>& wav) {
        constexpr uint64_t S = T - M;          // chunk stride in frames
        constexpr int      R = N_FFT - N_HOP;  // samples still open after each hop

        fftwf_complex* in  = fftwf_alloc_complex(N_BIN);
        float*         out = fftwf_alloc_real(N_FFT);
        std::vector<float> acc(N_FFT);         // windowed overlap-add of the signal
        std::vector<float> env(N_FFT);         // overlap-add of WIN^2 (true envelope)

        const auto plan = fftwf_plan_dft_c2r_1d(N_FFT, in, out, FFTW_ESTIMATE);

        const auto emit = [&](std::vector<float>& ch, const int64_t idx, const int k) {
            if (idx < 0 || idx >= static_cast<int64_t>(L_)) return;
            ch[idx] = env[k] > 1e-8f ? acc[k] / (env[k] * N_FFT) : 0.0f;
        };

        for (int c = 0; c < C; ++c) {
            std::ranges::fill(acc, 0.0f);
            std::ranges::fill(env, 0.0f);
            int64_t base = -N_FFT / 2;         // sample index of acc[0]; frame ta spans
                                               // [ta*N_HOP - N_FFT/2, ta*N_HOP + N_FFT/2)
                                               // == [base, base + N_FFT) exactly
            for (uint64_t ta = 0; ta < T_; ++ta) {
                // chunk selection: n2 owns frame ta; its first M frames (t2 < M) are
                // shared with chunk n2-1 (as local frame t2 + S) and mask-crossfaded
                const auto  n2 = std::min<uint64_t>(N_ - 1, ta / S);
                const auto  t2 = static_cast<int>(ta - n2 * S);
                const bool  bl = n2 > 0 && t2 < M;
                const float a  = bl ? t2 / (M - 1.0f) : 1.0f;

                for (int f = 0; f < N_BIN; ++f) {
                    const auto ks = (c * T_ + ta) * N_BIN + f;
                    const auto kc = n2 * N_BUF + (((t2 * N_BIN + f) * C + c) << 1);
                    float mr = a * buf[kc];
                    float mi = a * buf[kc + 1];
                    if (bl) {
                        const auto kp = (n2 - 1) * N_BUF + ((((t2 + S) * N_BIN + f) * C + c) << 1);
                        mr += (1.0f - a) * buf[kp];
                        mi += (1.0f - a) * buf[kp + 1];
                    }
                    in[f][0] = spec[ks][0] * mr - spec[ks][1] * mi;
                    in[f][1] = spec[ks][0] * mi + spec[ks][1] * mr;
                }

                fftwf_execute(plan);

                for (int i = 0; i < N_FFT; ++i) {
                    acc[i] += out[i] * WIN[i];
                    env[i] += WIN[i] * WIN[i];
                }

                // frame ta+1 first touches sample base + N_HOP, so [base, base + N_HOP)
                // is final: normalize, emit, slide the window by one hop
                for (int k = 0; k < N_HOP; ++k) emit(wav[c], base + k, k);
                std::memmove(acc.data(), acc.data() + N_HOP, R * sizeof(float));
                std::memmove(env.data(), env.data() + N_HOP, R * sizeof(float));
                std::fill_n(acc.data() + R, N_HOP, 0.0f);
                std::fill_n(env.data() + R, N_HOP, 0.0f);
                base += N_HOP;
            }

            for (int k = 0; k < R; ++k) emit(wav[c], base + k, k);
        }

        fftwf_destroy_plan(plan);
        fftwf_free(in);
        fftwf_free(out);
    }
}
